cmake_minimum_required(VERSION 3.20)
project(GaalopQraNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(QRA_RESOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor" CACHE PATH "Directory containing garamon and eigen-3.4.1 (defaults to bundled sources)")
set(QRA_QUBITS "2;3;4;5;6;7;8;9" CACHE STRING "Qubit libraries to build")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "QRA requires a 64-bit x86 toolchain")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Supported platforms are Windows and Linux x86_64")
endif()
if(NOT EXISTS "${QRA_RESOURCE_ROOT}/garamon/src/main.cpp")
  message(FATAL_ERROR "Set QRA_RESOURCE_ROOT to the supplied QRA_resource directory")
endif()
set(EIGEN_INCLUDE "${QRA_RESOURCE_ROOT}/eigen-3.4.1")
if(NOT EXISTS "${EIGEN_INCLUDE}/Eigen/Core")
  message(FATAL_ERROR "Missing Eigen headers: ${EIGEN_INCLUDE}/Eigen/Core")
endif()
file(GLOB GENERATOR_SOURCES "${QRA_RESOURCE_ROOT}/garamon/src/*.cpp")
add_executable(qra_generator ${GENERATOR_SOURCES})
target_include_directories(qra_generator PRIVATE "${EIGEN_INCLUDE}")
if(MSVC)
  target_compile_options(qra_generator PRIVATE /bigobj /O2)
endif()
# The upstream generator reads ../data relative to its working directory.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generator/build/output")
file(COPY "${QRA_RESOURCE_ROOT}/garamon/data" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/generator")
include(CTest)
foreach(N IN LISTS QRA_QUBITS)
  if(NOT N MATCHES "^[2-9]$")
    message(FATAL_ERROR "QRA_QUBITS must contain integers 2 through 9")
  endif()
  math(EXPR DIM "2 * ${N} + 2")
  math(EXPR LAST "${DIM} - 1")
  math(EXPR PHYSICAL "2 * ${N}")
  set(METRIC "")
  foreach(R RANGE 0 ${LAST})
    foreach(C RANGE 0 ${LAST})
      if(R EQUAL C)
        string(APPEND METRIC "1 ")
      else()
        string(APPEND METRIC "0 ")
      endif()
    endforeach()
    string(APPEND METRIC "\n")
  endforeach()
  set(BASIS "")
  foreach(I RANGE 1 ${PHYSICAL})
    # Prefix tokens so b1+b2 cannot collide with b12. Only C++ accessor names
    # change; the bit positions and GAALOP e1..e(2n),er1,er2 basis stay unchanged.
    string(APPEND BASIS "b${I} ")
  endforeach()
  string(APPEND BASIS "r1 r2")
  configure_file(qra.conf.in "conf/qra${N}.conf" @ONLY)
  set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generator/build/output/garamon_qra${N}")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/qra${N}.stamp"
    COMMAND "${CMAKE_COMMAND}" -DGENERATOR=$<TARGET_FILE:qra_generator>
      -DCONFIG=${CMAKE_CURRENT_BINARY_DIR}/conf/qra${N}.conf -DOUTPUT=${GENERATED}
      -P "${CMAKE_CURRENT_SOURCE_DIR}/generate.cmake"
    COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_CURRENT_BINARY_DIR}/qra${N}.stamp"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generator/build"
    DEPENDS qra_generator "${CMAKE_CURRENT_BINARY_DIR}/conf/qra${N}.conf" generate.cmake)
  add_custom_target(qra${N}_generate DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/qra${N}.stamp")
  configure_file(bridge.cpp.in "bridge${N}.cpp" @ONLY)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/bridge${N}.cpp"
    PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/qra${N}.stamp")
  add_library(Qra${N}Bridge SHARED "${CMAKE_CURRENT_BINARY_DIR}/bridge${N}.cpp")
  if(WIN32)
    set_target_properties(Qra${N}Bridge PROPERTIES PREFIX "")
  endif()
  if(MINGW)
    target_link_options(Qra${N}Bridge PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_dependencies(Qra${N}Bridge qra${N}_generate)
  target_include_directories(Qra${N}Bridge PRIVATE "${GENERATED}/src" "${EIGEN_INCLUDE}")
  if(MSVC)
    target_compile_options(Qra${N}Bridge PRIVATE /bigobj /O2)
  endif()
  if(WIN32)
    set(PLATFORM windows-x86_64)
  else()
    set(PLATFORM linux-x86_64)
  endif()
  install(TARGETS Qra${N}Bridge RUNTIME DESTINATION ${PLATFORM}/qra${N}
    LIBRARY DESTINATION ${PLATFORM}/qra${N})
  if(BUILD_TESTING)
    configure_file(smoke.cpp.in "smoke${N}.cpp" @ONLY)
    add_executable(qra${N}_smoke "${CMAKE_CURRENT_BINARY_DIR}/smoke${N}.cpp")
    if(MINGW)
      target_link_options(qra${N}_smoke PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    target_link_libraries(qra${N}_smoke PRIVATE Qra${N}Bridge)
    add_test(NAME qra${N}_products COMMAND qra${N}_smoke)
  endif()
endforeach()
