// === Basic Definitions ===
//  Imaginary unit
i = er1*er2;

// Witt basis
f1 = 0.5*(e1 + i*e3);
f1T = 0.5*(e1 - i*e3);
f2 = 0.5*(e2 + i*e4);
f2T = 0.5*(e2 - i*e4);

// Element I (Identity)
Id = f1*f1T*f2*f2T;

// === Ket Basis Vectors ===
ket00 = 1*Id;
ket01 = f2T*Id;
ket10 = f1T*Id;
ket11 = f1T*f2T*Id;

// === Quantum Gates ===

// Hadamard Gates
h1 = (1/sqrt(2))*(f1*f1T + f1 + f1T - f1T*f1);
h2 = (1/sqrt(2))*(f2*f2T + f2 + f2T - f2T*f2);

// NOT Gates (Pauli-X)
not1 = f1 + f1T;
not2 = f2 + f2T;

// Multi-Controlled Z (flips phase of |11...1>)
mcz = (f1*f1T + f1T*f1*(f2*f2T - f2T*f2));

// === Diffusion Operator ===
diffusion = h1*h2*not1*not2 * mcz * not2*not1*h2*h1;

// === Oracle Operator ===
oracle = (not2 * mcz * not2);

// === Grover Operator ===
grover = diffusion * oracle;

// Iterations needed: 1
full_operator = grover;

// === Algorithm Execution ===
// superposition
psi = h1*h2*Id;

res = full_operator * psi;

// Result
?res;
