// === Basic Definitions ===
//  Imaginary unit
i = er1*er2;

// Witt basis
f1 = 0.5*(e1 + i*e5);
f1T = 0.5*(e1 - i*e5);
f2 = 0.5*(e2 + i*e6);
f2T = 0.5*(e2 - i*e6);
f3 = 0.5*(e3 + i*e7);
f3T = 0.5*(e3 - i*e7);
f4 = 0.5*(e4 + i*e8);
f4T = 0.5*(e4 - i*e8);

// Element I (Identity)
Id = f1*f1T*f2*f2T*f3*f3T*f4*f4T;

// === Ket Basis Vectors ===
ket0000 = 1*Id;
ket0001 = f4T*Id;
ket0010 = f3T*Id;
ket0011 = f3T*f4T*Id;
ket0100 = f2T*Id;
ket0101 = f2T*f4T*Id;
ket0110 = f2T*f3T*Id;
ket0111 = f2T*f3T*f4T*Id;
ket1000 = f1T*Id;
ket1001 = f1T*f4T*Id;
ket1010 = f1T*f3T*Id;
ket1011 = f1T*f3T*f4T*Id;
ket1100 = f1T*f2T*Id;
ket1101 = f1T*f2T*f4T*Id;
ket1110 = f1T*f2T*f3T*Id;
ket1111 = f1T*f2T*f3T*f4T*Id;

// === Quantum Gates ===

// Hadamard Gates
h1 = (1/sqrt(2))*(f1*f1T + f1 + f1T - f1T*f1);
h2 = (1/sqrt(2))*(f2*f2T + f2 + f2T - f2T*f2);
h3 = (1/sqrt(2))*(f3*f3T + f3 + f3T - f3T*f3);
h4 = (1/sqrt(2))*(f4*f4T + f4 + f4T - f4T*f4);

// NOT Gates (Pauli-X)
not1 = f1 + f1T;
not2 = f2 + f2T;
not3 = f3 + f3T;
not4 = f4 + f4T;

// Multi-Controlled Z (flips phase of |11...1>)
mcz = (f1*f1T + f1T*f1*(f2*f2T + f2T*f2*(f3*f3T + f3T*f3*(f4*f4T - f4T*f4))));

// === Diffusion Operator ===
diffusion = h1*h2*h3*h4*not1*not2*not3*not4 * mcz * not4*not3*not2*not1*h4*h3*h2*h1;

// === Oracle Operator ===
oracle = (not2*not3*not4 * mcz * not4*not3*not2);

// === Grover Operator ===
grover = diffusion * oracle;

// Iterations needed: 2
full_operator = grover*grover;

// === Algorithm Execution ===
// superposition
psi = h1*h2*h3*h4*Id;

res = full_operator * psi;

// Result
?res;
